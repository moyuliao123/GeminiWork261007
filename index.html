async function autoParseLink() { const urlInput = document.getElementById('knowLinkInput'); const titleInput = document.getElementById('knowTitleInput'); const catInput = document.getElementById('knowCategoryInput'); const rawUrl = urlInput.value.trim(); if (!rawUrl) { alert('請先貼上網址連結！'); return; } // 1. 驗證 URL let parsedUrl; try { parsedUrl = new URL(rawUrl); } catch (e) { alert('請輸入包含 http:// 或 https:// 的有效網址！'); return; } // 2. 顯示讀取中狀態 titleInput.placeholder = '正在深入抓取真實網頁標題，請稍候...'; titleInput.value = ''; try { // 透過免費跨域代理抓取真實 HTML const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(rawUrl)}`; const response = await fetch(proxyUrl); const data = await response.json(); if (data && data.contents) { // 解析 HTML DOM const parser = new DOMParser(); const doc = parser.parseFromString(data.contents, 'text/html'); // 優先抓取 Open Graph 標題，其次抓取標準 const ogTitle = doc.querySelector('meta[property="og:title"]')?.getAttribute('content'); const docTitle = doc.querySelector('title')?.textContent; let finalTitle = (ogTitle || docTitle || '').trim(); // 清理網站後綴（例如「- 三采網路書店」、「| 博客來」） finalTitle = finalTitle .replace(/\s*[-|_–]\s*(三采文化|三采網路書店|博客來|誠品|Readmoo).*$/i, '') .trim(); if (finalTitle) { titleInput.value = finalTitle; } else { throw new Error('未找到標題'); } } else { throw new Error('無法讀取內容'); } } catch (err) { // 若代理超時或被網站阻擋，退回以網址路徑作為備用方案 console.warn('線上擷取失敗，改用網址備用解析：', err); let fallback = parsedUrl.pathname.split('/').filter(Boolean).pop() || parsedUrl.hostname; titleInput.value = decodeURIComponent(fallback).replace(/[-_]/g, ' '); titleInput.placeholder = '例如：福爾摩斯思考術、怪誕心理學觀察...'; } // 3. 領域自動推測 const fullText = (titleInput.value + ' ' + rawUrl).toLowerCase(); if (fullText.includes('推理') || fullText.includes('偵探') || fullText.includes('小說') || fullText.includes('mystery')) { catInput.value = '推理小說'; } else if (fullText.includes('心理') || fullText.includes('大腦') || fullText.includes('行為') || fullText.includes('思考')) { catInput.value = '心理學'; } else if (fullText.includes('程式') || fullText.includes('演算法') || fullText.includes('ai') || fullText.includes('科技')) { catInput.value = '程式科技'; } else if (!catInput.value) { catInput.value = '課外閱讀'; } }
